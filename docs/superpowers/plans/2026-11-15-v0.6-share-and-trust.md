# v0.6 分享入口与信任可见 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 App 接进 Android 分享动线，并把「无网络权限、纯本地、加密保护边界」的信任写清楚；同时把新包 PBKDF2 迭代提升到 600,000。

**Architecture:** 在 MainActivity 解析 `ACTION_SEND` / `ACTION_SEND_MULTIPLE` 的 URI，交给 MainViewModel 用现有 `UriUtils` / `CarrierCodec` 分类；纯分类逻辑放在可做 JVM 测试的 `ShareClassifier`。生成结果通过现有 FileProvider 分享。KDF 只改新包默认迭代数，格式布局不变。信任内容落在关于页、两份 README 和根目录 `SECURITY.md`。

**Tech Stack:** Kotlin、Jetpack Compose Material 3、AndroidX、kotlinx.coroutines/serialization、JUnit4、Android instrumentation tests。

**Spec:** `docs/superpowers/specs/2026-11-15-v0.6-share-and-trust-design.md`

## Global Constraints

- `minSdk = 29`，`targetSdk = 35`；不新增权限，尤其不得引入 `INTERNET`。
- 不改变 `tpWj` / `IFB2` 二进制布局；新加密包 PBKDF2 迭代 600,000，读取端继续接受 10,000–1,000,000。
- 版本号 `versionCode = 7`，`versionName = "0.6.0"`。
- 不做 F-Droid / IzzyOnDroid 元数据或上架。
- 只做本地提交，不 push、不开 PR。
- 新增用户可见文案必须同时补 `values/strings.xml` 与 `values-en/strings.xml`。

## Review Focus

- 只分享文本、没有 `EXTRA_STREAM`：必须提示「没有可读取的文件」，不能崩溃或清空现有页面。
- 分享大于 64 MB 的图片：不做隐藏内容探测，按普通载荷处理，避免 OOM；用户仍可之后手动提取。
- 连续两次分享：按 URI 去重合并，不能重复添加同一文件。
- 分享加密图片：能识别为含隐藏内容并弹出「提取 / 打包」选择；提取仍走密码流程。
- 生成结果分享时 FileProvider / 缓存写入失败：snackbar 提示，不崩溃。

```kotlin
    val packed = BundleCodec.pack(entries, compress = true, password = "pw")
    val iterations = ((packed[6].toInt() and 0xFF) shl 24) or
        ((packed[7].toInt() and 0xFF) shl 16) or

        (packed[9].toInt() and 0xFF)
    assertEquals(600_000, iterations)

@Test
fun `legacy 150000 iteration envelope still decrypts`() {
    val zip = zipOf("old.txt", "legacy payload".toByteArray(StandardCharsets.UTF_8))
    val envelope = legacyEncryptedEnvelope(zip, "pw", 150_000)
    val unpacked = BundleCodec.unpack(envelope, "pw")
    assertEquals(listOf("old.txt"), unpacked.map { it.path })
    assertArrayEquals("legacy payload".toByteArray(StandardCharsets.UTF_8), unpacked.single().bytes)
}

private fun zipOf(path: String, bytes: ByteArray): ByteArray =
    ByteArrayOutputStream().also { out ->
        ZipOutputStream(out).use { zip ->
            zip.putNextEntry(ZipEntry(path))
            zip.write(bytes)
            zip.closeEntry()
        }
    }.toByteArray()

/** Builds an IFB2 envelope using an arbitrary iteration count. */
private fun legacyEncryptedEnvelope(
    zipBytes: ByteArray,
    password: String,
    iterations: Int,
): ByteArray {
    val salt = ByteArray(16) { (it + 1).toByte() }
    val nonce = ByteArray(12) { (it + 21).toByte() }
    val header = ByteArrayOutputStream().also { out ->
        out.write("IFB2".toByteArray(StandardCharsets.US_ASCII))
        out.write(1)
        out.write(1)
        out.write(byteArrayOf(
            (iterations ushr 24).toByte(),
            (iterations ushr 16).toByte(),
            (iterations ushr 8).toByte(),
            iterations.toByte(),
        ))
        out.write(16)
        out.write(12)
        out.write(salt)
        out.write(nonce)
    }.toByteArray()
    val key = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256")
        .generateSecret(PBEKeySpec(password.toCharArray(), salt, iterations, 256))
        .encoded
    val cipher = Cipher.getInstance("AES/GCM/NoPadding")
    cipher.init(
        Cipher.ENCRYPT_MODE,
        SecretKeySpec(key, "AES"),
        GCMParameterSpec(128, nonce),
    )
    cipher.updateAAD(header)
    return header + cipher.doFinal(zipBytes)
}
```

```kotlin
data class SharedFile(
    val uri: String,
    val name: String,
    val mime: String?,
    val size: Long,
    val hasHiddenPayload: Boolean,
)

data class SharePlan(
    val packUris: List<String>,
    val extractUris: List<String>,
) {
    val needsChoice: Boolean get() = extractUris.isNotEmpty()
}

object ShareClassifier {
    const val MAX_INSPECT_BYTES = 64 * 1024 * 1024

    fun shouldInspect(size: Long): Boolean = size < 0 || size <= MAX_INSPECT_BYTES

    fun classify(files: List<SharedFile>): SharePlan {
        val unique = files.distinctBy { it.uri }
        return SharePlan(
            packUris = unique.map { it.uri },
            extractUris = unique.filter { it.hasHiddenPayload }.map { it.uri },
        )
    }
}
```

```kotlin
/** Reads at most [maxBytes]; returns null when the stream is larger. */
fun readAtMost(context: Context, uri: Uri, maxBytes: Int): ByteArray? {
    context.contentResolver.openInputStream(uri)?.use { input ->
        val out = java.io.ByteArrayOutputStream()
        val buffer = ByteArray(8192)
        var total = 0
        while (true) {
            val read = input.read(buffer)
            if (read < 0) break
            total += read
            if (total > maxBytes) return null
            out.write(buffer, 0, read)
        }
        return out.toByteArray()
    }
    return null
}
```

```kotlin
data class PendingShare(
    val extractSources: List<PickedFile>,
    val packSources: List<PickedFile>,
)

private val _activeTab = MutableStateFlow(0)
val activeTab: StateFlow<Int> = _activeTab.asStateFlow()

private val _pendingShare = MutableStateFlow<PendingShare?>(null)
val pendingShare: StateFlow<PendingShare?> = _pendingShare.asStateFlow()

fun selectTab(index: Int) {
    if (index in 0..2) _activeTab.value = index
}

fun handleSharedUris(uris: List<Uri>) {
    if (uris.isEmpty()) {
        showMessage(getApplication<Application>().getString(R.string.msg_share_no_files))
        return
    }
    viewModelScope.launch {
        val app = getApplication<Application>()
        val resolved = withContext(Dispatchers.IO) {
            uris.distinct().mapNotNull { uri ->
                runCatching { UriUtils.describe(app, uri) }.getOrNull()
            }
        }
        if (resolved.isEmpty()) {
            showMessage(app.getString(R.string.msg_share_no_files))
            return@launch
        }
        val shared = withContext(Dispatchers.IO) {
            resolved.map { file ->
                val inspect = file.mime?.startsWith("image/") == true &&
                    ShareClassifier.shouldInspect(file.size)
                val hasHidden = inspect && runCatching {
                    val bytes = UriUtils.readAtMost(
                        app,
                        file.uri,
                        ShareClassifier.MAX_INSPECT_BYTES,
                    )
                    bytes != null && CarrierCodec.extractItems(bytes).isNotEmpty()
                }.getOrDefault(false)
                SharedFile(
                    uri = file.uri.toString(),
                    name = file.name,
                    mime = file.mime,
                    size = file.size,
                    hasHiddenPayload = hasHidden,
                )
            }
        }
        val plan = ShareClassifier.classify(shared)
        val byUri = resolved.associateBy { it.uri.toString() }
        val incomingExtract = plan.extractUris.mapNotNull(byUri::get)
        val incomingPack = plan.packUris.mapNotNull(byUri::get)
        val current = _pendingShare.value
        if (plan.needsChoice || current != null) {
            // 对话框已存在时按 URI 去重合并，避免重复添加。
            _pendingShare.value = PendingShare(
                extractSources = (current?.extractSources.orEmpty() + incomingExtract)
                    .distinctBy { it.uri },
                packSources = (current?.packSources.orEmpty() + incomingPack)
                    .distinctBy { it.uri },
            )
        } else {
            addPayloadsDirect(incomingPack)
            selectTab(0)
            showMessage(app.getString(R.string.msg_share_added, incomingPack.size))
        }
    }
}

fun confirmShareExtract() {
    val pending = _pendingShare.value ?: return
    _pendingShare.value = null
    _extract.update {
        it.copy(
            sources = pending.extractSources,
            results = emptyList(),
            hasExtracted = false,
            needsPassword = false,
            passwordError = false,
        )
    }
    selectTab(1)
}

fun confirmSharePack() {
    val pending = _pendingShare.value ?: return
    _pendingShare.value = null
    addPayloadsDirect(pending.packSources)
    selectTab(0)
}

fun dismissPendingShare() {
    _pendingShare.value = null
}

private fun addPayloadsDirect(files: List<PickedFile>) {
    _pack.update { state ->
        val existing = state.payloads.map { it.uri }.toSet()
        state.copy(payloads = state.payloads + files.filterNot { it.uri in existing })
    }
}
```

`runCatching { file.mime... }` 里的 `readBytes` 对未知大小或超大文件不做探测。`file.mime` 为 null 时按非图片处理，不探测。

```kotlin
import android.content.Intent
import android.net.Uri
import android.os.Build
import androidx.activity.viewModels

class MainActivity : AppCompatActivity() {
    private val viewModel: MainViewModel by viewModels()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent { AppRoot(viewModel = viewModel) }
        if (savedInstanceState == null) handleShareIntent(intent)
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        setIntent(intent)
        handleShareIntent(intent)
    }

    private fun handleShareIntent(intent: Intent?) {
        if (intent == null) return
        if (intent.action != Intent.ACTION_SEND && intent.action != Intent.ACTION_SEND_MULTIPLE) return
        val uris = streamUris(intent)
        if (uris.isEmpty()) {
            viewModel.showMessage(getString(R.string.msg_share_no_files))
            return
        }
        viewModel.handleSharedUris(uris)
    }

    @Suppress("DEPRECATION")
    private fun streamUris(intent: Intent): List<Uri> {
        val list = if (Build.VERSION.SDK_INT >= 33) {
            intent.getParcelableArrayListExtra(Intent.EXTRA_STREAM, Uri::class.java)
        } else {
            intent.getParcelableArrayListExtra(Intent.EXTRA_STREAM)
        }
        val single = if (Build.VERSION.SDK_INT >= 33) {
            intent.getParcelableExtra(Intent.EXTRA_STREAM, Uri::class.java)
        } else {
            intent.getParcelableExtra(Intent.EXTRA_STREAM)
        }
        val clip = intent.clipData?.let { data ->
            (0 until data.itemCount).mapNotNull { data.getItemAt(it).uri }
        }.orEmpty()
        return (list.orEmpty() + listOfNotNull(single) + clip).distinct()
    }
}
```

```kotlin
package com.imagefolder.open.ui

import android.content.Context
import android.content.Intent
import android.net.Uri
import androidx.core.content.FileProvider
import com.imagefolder.open.GeneratedResult

internal fun writeShareCache(context: Context, bytes: ByteArray, fileName: String): Uri {
    val directory = java.io.File(context.cacheDir, "shared").apply { mkdirs() }
    val file = java.io.File(directory, fileName.replace('/', '_'))
    file.writeBytes(bytes)
    return FileProvider.getUriForFile(context, "${context.packageName}.fileprovider", file)
}

internal fun shareGeneratedImage(context: Context, result: GeneratedResult) {
    val uri = writeShareCache(context, result.bytes, result.fileName)
    val send = Intent(Intent.ACTION_SEND).apply {
        type = result.mime
        putExtra(Intent.EXTRA_STREAM, uri)
        addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
    }
    context.startActivity(Intent.createChooser(send, null))
}
```
